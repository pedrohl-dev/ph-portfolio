import random

def escolher_com_peso(opcoes, pesos):
    if len(opcoes) != len(pesos):
        raise ValueError("Opções e pesos devem ter o mesmo tamanho")

    total = sum(pesos)
    r = random.uniform(0, total)
    acumulado = 0

    for opcao, peso in zip(opcoes, pesos):
        acumulado += peso
        if r <= acumulado:
            return opcao
